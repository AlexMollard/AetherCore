cmake_minimum_required(VERSION 4.0)
project(AetherBrowserCppProbe LANGUAGES C CXX)

if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "This probe requires the Emscripten toolchain")
endif()
if(NOT EXISTS "${AETHER_BOX2D_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "AETHER_BOX2D_SOURCE must name the existing Box2D 3.1.1 source directory")
endif()
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
add_subdirectory("${AETHER_BOX2D_SOURCE}" "${CMAKE_BINARY_DIR}/box2d")
add_executable(cpp-probe CppProbe.cpp)
target_link_libraries(cpp-probe PRIVATE box2d)
target_compile_options(cpp-probe PRIVATE -fno-exceptions)
target_link_options(cpp-probe PRIVATE -fno-exceptions -sASSERTIONS=2 -sEXIT_RUNTIME=1 -sENVIRONMENT=web,node
    "--shell-file=${CMAKE_CURRENT_SOURCE_DIR}/shell.html")
set_target_properties(cpp-probe PROPERTIES SUFFIX ".html")
