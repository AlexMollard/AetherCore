cmake_minimum_required(VERSION 3.24)
project(MeowCore LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ---------------------------------------------------------------------------
# 1. Dependency Management (FetchContent)
# ---------------------------------------------------------------------------
include(FetchContent)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/CMake")

# Define our external deps
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw GIT_TAG 3.4)
FetchContent_Declare(glm  GIT_REPOSITORY https://github.com/g-truc/glm  GIT_TAG 1.0.3)
FetchContent_Declare(vk-bootstrap GIT_REPOSITORY https://github.com/charles-lunarg/vk-bootstrap GIT_TAG v1.4.341)
FetchContent_Declare(VMA GIT_REPOSITORY https://github.com/GPUOpen-LibrariesAndSDKs/VulkanMemoryAllocator.git GIT_TAG v3.3.0)

FetchContent_MakeAvailable(glfw glm vk-bootstrap VMA)

# ---------------------------------------------------------------------------
# 2. The Core Render Engine (Library)
# ---------------------------------------------------------------------------
# We isolate the engine so the App just "uses" it.
add_library(Engine STATIC)

target_sources(Engine PRIVATE
    src/engine/BindlessContract.hpp
    src/engine/BindlessContract.cpp
    src/engine/BindlessManager.hpp
    src/engine/BindlessManager.cpp
    src/engine/CommandRecorder.hpp
    src/engine/CommandRecorder.cpp
    src/engine/CrashHandler.hpp
    src/engine/CrashHandler.cpp
    src/engine/GraphicsPipeline.hpp
    src/engine/GraphicsPipeline.cpp
    src/engine/LogCategory.hpp
    src/engine/Logger.hpp
    src/engine/Logger.cpp
    src/engine/MeowExceptions.hpp
    src/engine/MeowCore.cpp
    src/engine/MeowCore.hpp
    src/engine/Mesh.hpp
    src/engine/Mesh.cpp
    src/engine/PrimitiveMeshes.hpp
    src/engine/PrimitiveMeshes.cpp
    src/engine/PassResourceCompiler.hpp
    src/engine/PassResourceCompiler.cpp
    src/engine/ResourcePool.hpp
    src/engine/ResourcePool.cpp
    src/engine/RenderGraph.cpp
    src/engine/RenderGraph.hpp
    src/engine/RenderQueue.hpp
    src/engine/RenderQueue.cpp
    src/engine/Scene.hpp
    src/engine/Scene.cpp
    src/engine/Swapchain.hpp
    src/engine/Swapchain.cpp
    src/engine/UniqueBuffer.hpp
    src/engine/UniqueBuffer.cpp
    src/engine/UniqueImage.hpp
    src/engine/UniqueImage.cpp
    src/engine/VulkanContext.hpp
    src/engine/VulkanContext.cpp
    src/engine/VulkanUtils.hpp
    src/engine/VmaImplementation.cpp
    src/engine/Window.hpp
    src/engine/Window.cpp
)

target_sources(Engine PRIVATE
    src/engine/io/IFileBackend.hpp
    src/engine/io/FileGlobOptions.hpp
    src/engine/io/DirectoryBackend.hpp
    src/engine/io/DirectoryBackend.cpp
    src/engine/io/FileRequest.hpp
    src/engine/io/FileRequest.cpp
    src/engine/io/IOThread.hpp
    src/engine/io/IOThread.cpp
    src/engine/io/FileSystem.hpp
    src/engine/io/FileSystem.cpp
)

# Find Vulkan on the system
find_package(Vulkan REQUIRED)

include(SlangShaders)

target_link_libraries(Engine PUBLIC 
    Vulkan::Vulkan
    glfw
    glm::glm
    vk-bootstrap
)

if (WIN32)
    target_link_libraries(Engine PUBLIC dbghelp)
endif()

# Tell the engine where its own headers are
target_include_directories(Engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/engine)
target_include_directories(Engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/engine/io)
target_include_directories(Engine PUBLIC ${vma_SOURCE_DIR}/include)

if (MSVC)
    target_compile_options(Engine PRIVATE /Zc:preprocessor)
endif()

# ---------------------------------------------------------------------------
# 3. The Application (Executable)
# ---------------------------------------------------------------------------
add_executable(App)

target_sources(App PRIVATE
    src/main.cpp
    src/app/Application.cpp
    src/app/Application.hpp
    src/app/AppLayer.hpp
    src/app/LayerStack.cpp
    src/app/LayerStack.hpp
    src/app/layers/SandboxLayer.cpp
    src/app/layers/SandboxLayer.hpp
)

target_include_directories(App PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/app)

if (MSVC)
    target_compile_options(App PRIVATE /Zc:preprocessor)
endif()

# The App "inherits" Vulkan, GLFW, and GLM automatically from the Engine target!
target_link_libraries(App PRIVATE Engine)

meowcore_enable_slang_shader_compilation(App)
