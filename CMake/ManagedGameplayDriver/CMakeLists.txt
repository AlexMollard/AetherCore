cmake_minimum_required(VERSION 3.25)
project(AetherManagedGameplay LANGUAGES NONE)
foreach(required AETHER_GAMEPLAY_PROJECT AETHER_DOTNET_EXECUTABLE AETHER_MANAGED_ARTIFACTS AETHER_MANAGED_OUTPUT)
    if(NOT ${required})
        message(FATAL_ERROR "Missing ${required}")
    endif()
endforeach()
if(NOT EXISTS "${AETHER_GAMEPLAY_PROJECT}")
    message(FATAL_ERROR "The gameplay project is missing")
endif()
add_custom_target(AetherManagedGameplay
    COMMAND "${CMAKE_COMMAND}" "-DAETHER_DOTNET_EXECUTABLE=${AETHER_DOTNET_EXECUTABLE}"
        "-DAETHER_GAMEPLAY_PROJECT=${AETHER_GAMEPLAY_PROJECT}" "-DAETHER_MANAGED_ARTIFACTS=${AETHER_MANAGED_ARTIFACTS}"
        "-DAETHER_MANAGED_OUTPUT=${AETHER_MANAGED_OUTPUT}" -P "${CMAKE_CURRENT_LIST_DIR}/Build.cmake"
    VERBATIM USES_TERMINAL)
